\begin{abstract}
בעבודה הקודמת \emph{«עיוות אפיסטמי של הישגיות»} (\ltr{v3.2}) הוכח פורמלית כי הצירוף של מלכודת בולעת שאינה יעד ($G_1$), אי-הבחנה ליניארית בין מצבים ($G_2$) ומדידה מתערבת ($G_4$) שולל הישגיות מובטחת בכל אופק סופי $k \in \mathbb{N}$, תוך אי-פריקות מלאה לכל תת-קבוצה ממש של התחומים. ואולם בגרסה \ltr{v3.2} ההתאמה הבין-תחומית (\code{InterfaceGlue}) סופקה ידנית. בטרום-פרסום זה (\ltr{v4.0}), החיבור $\gamma = (\sigma_{12}, \sigma_{13})$ הופך ל\emph{אובייקט חיפוש בלתי ידוע} מעל מרחבי מצבים זרים בזוגות $S_1, S_2, S_3$ בעלי תמורות מזהים בלתי תלויות, והתהליך מתפרק לשני פלטים אורתוגונליים: אישור אי-פריקות משותפת (\code{CrossVerdict}) וסינתזה של עקיפה בטוחה בתפריט מורשה מבלי לשנות את המשימה המקורית (\code{BypassVerdict}). אנו מציגים גרעין פורמלי נקי ב-\ltr{Lean 4.36.0} (\ltr{18/18} הצהרות מאומתות, 0 ייבואים חיצוניים, \ltr{0} \code{sorryAx}, \ltr{0} \code{ofReduceBool}, \ltr{0} \code{Classical.choice}, מהן \ltr{15/18} ללא אקסיומות כלל; \ltr{SHA-256} \code{1bf6ce8a...}), מאמת אפס-אמון (\ltr{zero-trust}) מבודד פיזית, סדרת אמות מידה \ltr{X0--X4b} (כולל תת-אוטומט טבעתי בן 6 תאים של כלל וולפרם 110 על $\mathbb{Z}/6\mathbb{Z}$ והעברה למלכודות \ltr{SCC} מחזוריות עם פורטים מעל $\mathbb{F}_3^2$) וכן \ltr{12/12} בקרות שליליות מחייבות.
\end{abstract}

\section{מאימות חיבור לגילוי אוטונומי}

משפט בלקוול הקלאסי להשוואת ניסויים~\cite{blackwell1953,howard1966} מבטיח ערך אי-שלילי למידע חינמי ($\mathrm{VoI} \ge 0$) כל עוד התצפית אינה משנה את דינמיקת המערכת. בתכנון אפיסטמי דטרמיניסטי עם גשושים מתערבים~\cite{fagin1995,vanditmarsch2007,kaelbling1998,strand2026v32} עיקרון זה נשבר מבנית: צירוף של מלכודת בלתי הפיכה ($D_1$), הטלה ליניארית מאבדת מידע ($D_2$) וגשוש מעוות-מצב ($D_3$) שולל הישגיות מובטחת לכל $k \in \mathbb{N}$, בעוד הסרת כל תחום יחיד מחזירה אסטרטגיה מנצחת.

המעבר הארכיטקטוני ב-\ltr{v4.0} מחליף חיבור ידני בסינתזה אוטונומית ואימות חיצוני בלתי תלוי:
\begin{align}\label{eq:pipeline}
\text{\ltr{Shift 1 (DiscoverGlue):}}\quad &
(D_1, D_2, D_3) \xrightarrow{\;\mathrm{DiscoverGlue}\;} \bigl(\gamma^\star,\, \mathcal{L},\, \{T_{C_1}, T_{C_2}, T_{C_4}\}\bigr), \\
\text{\ltr{Shift 2 (DiscoverBypass):}}\quad &
(\gamma^\star, \mathcal{M}) \xrightarrow{\;\mathrm{DiscoverBypass}\;} \bigl(\rho^\star,\, T_{\mathrm{bypass}} \;\text{\ltr{or}}\; \{\mathcal{L}_\sigma\}_{\sigma \in \mathrm{Bij}}\bigr).
\end{align}
שלוש ספריות קפואות $(D_1, D_2, D_3)$ ניתנות מעל \emph{מרחבי מצבים אטומים וזרים בזוגות}:
\[
S_1 \cap S_2 = \varnothing, \qquad S_1 \cap S_3 = \varnothing, \qquad S_2 \cap S_3 = \varnothing.
\]
מנוע הגילוי אינו יכול להסתמך על שמות מצבים משותפים; עליו לסנתז במפורש התאמות חד-חד-ערכיות ועל $\sigma_{12}: S_1 \to S_2$ ו-$\sigma_{13}: S_1 \to S_3$, ולהפיק תעודת \ltr{JSON} עצמאית הנבדקת קדימה בלבד בידי מאמת אפס-אמון נפרד.

\takeaway{ההפרדה המרכזית ב-\ltr{v4.0}}{אישור חסימה בין-תחומית אי-פריקה (\code{IRREDUCIBLE}) אינו גורר קיום עקיפה בטוחה (\code{BYPASSABLE}). יתרה מזו, צמצום קבוצת המצבים ההתחלתית $B_0$ (\code{RestrictScope}) פותר משימה \emph{אחרת} ומופרד פורמלית מעקיפות מורשות (\code{RefinePort}, \code{ReplaceProbe}) המשמרות את $B_0$ ואת קבוצת היעדים $G$.}

\section{המודל הפורמלי ו-18 הצהרות מאומתות ב-Lean 4.36.0}

הגרעין הפורמלי מצוי בקובץ \code{lean/certified_structural_discovery_v4_0.lean} (909 שורות, 40,346 בתים, 0 ייבואים חיצוניים, הידור נקי ב-\ltr{Lean 4.36.0}, \ltr{SHA-256} \code{1bf6ce8aedabb3ce...}).

\subsection{הישגיות אמונה בטוחה ונאותות A4}

עולם דטרמיניסטי עם פורט פסיבי וגשוש מתערב מוגדר כחמישייה $W = (\delta, G, \pi, U_p, Y_p)$ מעל קבוצת המצבים $S$, כאשר $\delta: S \times \mathrm{Act} \to S$, $G \subseteq S$, $\pi: S \to \mathrm{Obs}$, $U_p: S \to S$ ו-$Y_p: S \to \mathrm{PObs}$. עוקבי אמונה פעילים לאחר סינון מצבי יעד מוגדרים כך:
\begin{align*}
\mathrm{actSucc}(W, B, a, o) &= \{ s' \in S \setminus G \mid \exists s \in B,\; \delta(s, a) = s' \land \pi(s') = o \}, \\
\mathrm{probeSucc}(W, B, y)  &= \{ s' \in S \setminus G \mid \exists s \in B,\; U_p(s) = s' \land Y_p(s) = y \}.
\end{align*}
נקודה מהותית היא ש-$\mathrm{probeSucc}$ אוכפת \emph{סמנטיקת תצפית קדם-שינוי}: הגשוש מחזיר את התצפית $Y_p(s)$ של המצב המקורי לפני העיוות $U_p(s)$ (מסונכרן בין \ltr{Lean}, מנוע הגילוי והמאמת, ונבדק בבקרת הרגרסיה \code{nc11}). הפרדיקט האינדוקטיבי $\mathrm{SafeBeliefReach}(W, k, B)$ מבטא קיום תוכנית מנצחת בעומק $\le k$, ו-$\mathrm{SafeBeliefReachInit}(W, k, B_0) \equiv \forall o,\; \mathrm{SafeBeliefReach}(W, k, B_0 \cap \pi^{-1}(o) \setminus G)$ משקלל את הפיצול הפסיבי בזמן $t=0$.

\begin{definition}[אזור הפסד סגור]\label{def:clr}
משפחת אמונות $\mathcal{L} \subseteq \mathcal{P}(S)$ נקראת \emph{אזור הפסד סגור} (\code{IsClosedLosingRegion}) אם:
(i) לכל $B \in \mathcal{L}$ קיים $s \in B \setminus G$;
(ii) לכל $B \in \mathcal{L}$ וכל $a \in \mathrm{Act}$ קיים $o \in \mathrm{Obs}$ כך ש-$\mathrm{actSucc}(W,B,a,o) \in \mathcal{L}$;
(iii) לכל $B \in \mathcal{L}$ קיים $y \in \mathrm{PObs}$ כך ש-$\mathrm{probeSucc}(W,B,y) \in \mathcal{L}$.
\end{definition}

\begin{theorem}[נאותות A4 לכל האופקים, משפטי 1--2 ב-Lean]\label{thm:a4}
אם $\mathcal{L}$ הוא אזור הפסד סגור ב-$W$ וקיים $o \in \mathrm{Obs}$ כך ש-$\mathrm{initBranch}(W, B_0, o) \in \mathcal{L}$, אז לכל $k \in \mathbb{N}$ מתקיים $\neg\mathrm{SafeBeliefReachInit}(W, k, B_0)$.
\end{theorem}

משפט~\ref{thm:a4} הופך את אזור ההפסד הסופי $\mathcal{L}$ שמפיק פותר נקודת השבת המינימלית \code{A4} להוכחת אי-אפשרות לכל $k \in \mathbb{N}$, הניתנת לבדיקה בסיבוכיות $O(|\mathcal{L}|\cdot|\mathrm{Act}|\cdot|S|)$ ללא חיפוש.

\subsection{שלמות הצטרפות מבנית (H16) ואקוויוואריאנטיות תלת-כיוונית (H18)}

עבור תחומים בלתי תלויים $D_1 = (\delta, G)$ על $S_1$, $D_2 = (\pi)$ על $S_2$ ו-$D_3 = (U_p, Y_p)$ על $S_3$, חיבור $\gamma = (\sigma_{12}, \sigma_{13}, T_1, s_0, s_1, t)$ מגדיר בייקציות מפורשות $\sigma_{12}: S_1 \to S_2$ ו-$\sigma_{13}: S_1 \to S_3$, מלכודת סגורה לא ריקה $T_1 \subseteq S_1 \setminus G$ (מלכודת נקודתית $|T_1|=1$ או רכיב קשיר היטב מחזורי $|T_1| \ge 2$) וזוג מתנגש $s_0 \ne s_1$ שבו $t \in \{s_0, s_1\}$ מועבר אל $T_1$ על ידי הגשוש.

\begin{theorem}[שלמות אינדקסים אינווריאנטיים לשמות H16, משפטים 3 ו-9 ב-Lean]\label{thm:h16}
כל חיבור תקין מבנית $\gamma$ (הן עבור מלכודות נקודתיות \code{ValidGlue} והן עבור מלכודות קבוצה \code{ValidSetGlue}) מקיים בהכרח את פרדיקטי האינדקס $I_1(D_1)$, $I_2(D_2)$ ו-$I_3(D_3)$. לפיכך ההצטרפות המבנית \code{A1} במצב מלא אינה מאבדת אף חיבור תקין מהחיפוש הממצה \code{A0} ($\mathrm{Accepted}(A_1) = \mathrm{Accepted}(A_0)$).
\end{theorem}

\begin{theorem}[אקוויוואריאנטיות תלת-כיוונית בלתי תלויה H18, משפט 4 ו-transport\_validGlue]\label{thm:h18}
לכל שלוש בייקציות שינוי-שמות \emph{בלתי תלויות} $p_1: S_1 \to S'_1$, $p_2: S_2 \to S'_2$, $p_3: S_3 \to S'_3$, ההעברה $\gamma' = (p_2 \circ \sigma_{12} \circ p_1^{-1},\; p_3 \circ \sigma_{13} \circ p_1^{-1},\; \dots)$ משמרת הן את \code{ValidGlue} והן את $\mathrm{SafeBeliefReachInit}(W_\gamma, k, B_0) \iff \mathrm{SafeBeliefReachInit}(W'_{\gamma'}, k, B_0 \circ p_1^{-1})$.
\end{theorem}

\subsection{עקיפה הוגנת (H14), נאותות אוניברסלית של UNBYPASSABLE והעברה (H19--H20)}

נגדיר $\mathrm{SolveOriginalTask}(W, B_0) \equiv \exists k,\; \mathrm{SafeBeliefReachInit}(W, k, B_0)$ וכן $\mathrm{SolveRestrictedTask}(W, B_0) \equiv \exists B_{\mathrm{sub}} \subseteq B_0,\; (B_{\mathrm{sub}} \ne \varnothing) \land \mathrm{SolveOriginalTask}(W, B_{\mathrm{sub}})$.
\begin{itemize}
\item \textbf{הפרדה קשיחה (\ltr{H14}, משפטים 5--6):} בעד בן 4 מצבים \code{witnessEDR4} מוכח $\mathrm{SolveRestrictedTask} \land \neg\mathrm{SolveOriginalTask}$: צמצום ל-$\{s_0\}$ מנצח בצעד אחד, בעוד המשימה המקורית על $\{s_0, s_1\}$ בלתי אפשרית לכל $k \in \mathbb{N}$.
\item \textbf{נאותות אוניברסלית ל-\ltr{UNBYPASSABLE} (משפטים \ltr{7a--7b}):} שלילת פורט חלופי $D'_2$ או גשוש $D'_3$ דורשת משפחת אזורי הפסד סגורים $\{\mathcal{L}_\sigma\}$ על פני \emph{כל} הבייקציות $\sigma \in \mathrm{Bij}(S_1, S'_2)$ (או $\mathrm{Bij}(S_1, S'_3)$), וחוסמת זיוף כמת קיומי.
\item \textbf{הפרדת מלכודות \ltr{SCC} מחזוריות (\ltr{H20}, משפטים \ltr{8a--8b}):} העד בן 5 המצבים \code{witnessCyclic5} בעל מחזור-2 בולע $q_0 \leftrightarrow q_1$ נטול מלכודות נקודתיות ($\neg\mathrm{HasPointTrap}$), אך מוכח כבלתי אפשרי לכל $k \in \mathbb{N}$ באמצעות אותו גרעין \code{IsClosedLosingRegion} ללא כל שינוי.
\item \textbf{שער תנאי-קדם ואמורטיזציה מותנית (\ltr{H19}, \ltr{B6}, משפטים 10--11):} משפט 10 (\code{transferMethod_blocks_mismatched_scope}) מאמת את טבלת ההחלטה הסופית של שער התנאים (חסימת \code{pointSinkF2} על \ltr{SCC} מחזוריים ו-$\mathbb{F}_3$ ואישור \code{quotientSCCMod}), ומשפט 11 (\code{transferMethod_stream_amortization}) מוכיח $c_s + (k-1)c_m < k c_s$ לכל $c_m < c_s$ ו-$k \ge 2$ תחת מודל עלות מוצהר $(c_s, c_m)$.
\end{itemize}

\section{ארכיטקטורת הליבה המבצעת ומאמת אפס-אמון}

המימוש ב-\ltr{Python 3} (ספרייה תקנית בלבד) מופרד פיזית לשני מודולים:
\begin{enumerate}
\item \textbf{מנוע הגילוי (\code{certified_structural_discovery_v4_0.py}):} מממש חיפוש ממצה של כל זוגות הבייקציות \code{A0} ($\mathrm{Bij}(S_1,S_2)\times\mathrm{Bij}(S_1,S_3)$), הצטרפות מבנית אינווריאנטית לשמות \code{A1} ($I_1 \bowtie I_2 \bowtie I_3$ עם רישום מלא של עלות האינדוקס והזיכרון ב-\code{CostVector}), פותר נקודת שבת \code{A4} (המפיק \code{PlanTree} או \code{ClosedLosingRegion}), סינתזת עקיפה \code{DiscoverBypass} והעברת אסימוני שיטה \code{MethodToken}.

\paragraph{תת-אוטומט טבעתי של כלל וולפרם 110 (\code{X3}).}
המעברים מ-$c_0 = (1,1,0,0,1,0)$, $c_1 = (0,0,1,0,1,0)$ ומהוואקום $q_{\mathrm{vac}} = (0,0,0,0,0,0)$ מחושבים ישירות בשני צעדים סינכרוניים של כלל וולפרם 110~\cite{wolfram1983,cook2004} על $\mathbb{Z}/6\mathbb{Z}$ לאחר פולס הדלקת סיבית בתא 0 או 3: פולס בתא 3 מאיין את $c_0$ אל $q_{\mathrm{vac}}$ ופולס בתא 0 מאיין את $c_1$ אל $q_{\mathrm{vac}}$, ואילו הפולסים ההפוכים מובילים למצבי היעד החיים $g_0 = (0,1,1,1,0,0)$ ו-$g_1 = (1,1,1,1,1,0)$. מצבי היעד הבולעים $g_0, g_1$ ומצב העזר $c_{\mathrm{aux}}$ סוגרים את תת-האוטומט הסופי בן 6 המצבים; אין טענה לגבי אי-הפיכות על סריג אינסופי.

\paragraph{העברה מאושרת ל-\ltr{SCC} מחזוריים ול-$\mathbb{F}_3^2$ (\ltr{X4a--X4b}, \code{B6}).}
